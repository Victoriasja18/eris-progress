import importlib

def collect_coords(genome, target_db, label):
    rows = []
    with Pipeline(target_db=target_db) as pipeline:
        for locus in pipeline(genome):
            target = locus.targets[0] if locus.targets else None
            num_segments = len(target.segments) if target else 0

            # Extract each segment's coordinates separately
            if target:
                for seg in target.segments:
                    rows.append({
                        "label":        label,
                        "locus_id":     locus.id,
                        "target":       target.id,
                        "is_stitched":  num_segments > 1,
                        "contig":       seg.contig,
                        "seg_start":    seg.start,
                        "seg_end":      seg.end,
                        "strand":       seg.strand,
                        "seg_length":   seg.end - seg.start,
                    })
    return pd.DataFrame(rows)

Find how many locations hits

def intersect_loc(df1, df2, buffer):
    hits = 0
    matched = []
    
    for _, r1 in df1.iterrows():
        for _, r2 in df2.iterrows():
            if r1["transcript"] != r2["transcript"]:
                continue
            # Check if positions overlap with buffer
            if (r1["chromStart"] - buffer) <= r2["chromEnd"] and (r1["chromEnd"] + buffer) >= r2["chromStart"]:
                hits += 1
                matched.append({
                    "transcript": r1["transcript"],
                    "start_1":   r1["chromStart"],
                    "end_1":     r1["chromEnd"],
                    "start_2":   r2["chromStart"],
                    "end_2":     r2["chromEnd"],
                    "buffer":    buffer,
                })
    
    return hits, pd.DataFrame(matched)

# Run for each buffer
for buf in [10, 15, 25, 50, 100]:
    hits, df_matched = intersect_loc(comp_loc, ann_bed, buf)
    print(f"Buffer {buf:>3} bp → {hits} hits")
