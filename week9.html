from Bio.SeqRecord import SeqRecord
from Bio.Seq import Seq
from Bio import SeqIO

import matrix as mt

gt = mt.format_gt("../../data/complete_ec958_IS.reanotated.sorted.bed")

reference = SeqIO.to_dict(SeqIO.parse("../../data/EC958_complete_genome.fasta", "fasta"))

records = []
for _, row in gt.iterrows():
    seq = reference[row['transcript']].seq[row['chromStart']:row['chromEnd']]
    if row.get('strand') == '-':
        seq = seq.reverse_complement()

    record = SeqRecord(
        Seq(str(seq)),
        id=f"{row['name']}_{row['repeats']}",
        description=f"{row['transcript']}:{row['chromStart']}-{row['chromEnd']}"
    )
    records.append(record)

SeqIO.write(records, "../../data/gt_IS_nc_new.fasta", "fasta")

import IS_dict as isd

rows = isd.load_is_table()
name_to_synonyms, synonym_to_name = isd.build_synonym_dicts(rows)

combine["synonyms"] = combine["exp_name"].apply(
lambda n: isd.get_synonyms(n, name_to_synonyms, synonym_to_name) or pd.NA)

# recalculate the precision recall 
FP, TP, FN = 0, 0, 0
fp_list = {}

for i, row in combine.iterrows():
    target = row["target_name"]
    expected = row["exp_name"]
    synonyms = row["synonyms"]

    # synonyms may be pd.NA (no synonyms found) -- guard before using `in`
    synonym_match = False
    if isinstance(synonyms, (list, set, tuple)):
        synonym_match = target in synonyms

    if target == expected or synonym_match:
        TP += 1
    else:
        FP += 1
        fp_list[target] = expected

# False negatives
gt_keys = set(gt_n["name"])
found_names = set()
for _, row in combine.iterrows():
    found_names.add(row["target_name"])
    found_names.add(row["exp_name"])
    if isinstance(row["synonyms"], (list, set, tuple)):
        found_names.update(row["synonyms"])

missing_keys = gt_keys - found_names
FN = len(missing_keys)

print(f'Total True positive is: {TP} \nTotal False positive is: {FP} \nTotal False negative is: {FN}')
print(fp_list)
print("FN list:", missing_keys)
