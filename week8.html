def plot_swimlane(gt_dict, figsize=(16, None)):
    families = sorted(gt_dict.keys())
    n = len(families)
    height_per_row = 0.4
    figsize = (figsize[0], max(3, n * height_per_row + 1)) if figsize[1] is None else figsize

    palette = cm.get_cmap("tab20", n)
    color_map = {fam: palette(i) for i, fam in enumerate(families)}

    fig, ax = plt.subplots(figsize=figsize)
    bar_height = 0.5

    all_starts, all_ends = [], []

    # alternating row background bands, one per family lane
    for row_idx, fam in enumerate(families):
        if row_idx % 2 == 0:
            ax.axhspan(row_idx - 0.5, row_idx + 0.5, facecolor="grey", alpha=0.07, zorder=0)

    for row_idx, fam in enumerate(families):
        for start, end, strand in gt_dict[fam]:
            all_starts.append(start)
            all_ends.append(end)
            ax.add_patch(patches.Rectangle(
                (start, row_idx - bar_height / 2), end - start, bar_height,
                facecolor=color_map[fam], edgecolor="black", linewidth=0.5, zorder=2
            ))
            mid = (start + end) / 2
            marker = ">" if strand == "+" else "<"
            ax.plot(mid, row_idx, marker=marker, color="black", markersize=5, zorder=3)

    ax.set_yticks(range(n))
    ax.set_yticklabels(families, fontsize=8)
    ax.set_xlim(min(all_starts) - 5000, max(all_ends) + 5000)
    ax.set_xlabel("Genomic position (bp)")
    ax.set_title("IS element instances by family")

    # horizontal gridline at every family row boundary
    for row_idx in range(n + 1):
        ax.axhline(row_idx - 0.5, color="grey", linewidth=0.5, alpha=0.4, zorder=1)

    # vertical gridlines at regular genomic intervals
    ax.grid(axis="x", linestyle="--", alpha=0.3, zorder=1)
    ax.set_axisbelow(True)

    plt.tight_layout()
    plt.show()

import plotly.graph_objects as go
import matplotlib.cm as cm
import matplotlib.colors as mcolors

def using_plotly(gt_dict):
    families = sorted(gt_dict.keys())
    n = len(families)
    palette = cm.get_cmap("tab20", n)
    color_map = {fam: mcolors.to_hex(palette(i)) for i, fam in enumerate(families)}
    fam_to_row = {fam: i for i, fam in enumerate(families)}

    fig = go.Figure()

    # alternating row bands
    for i, fam in enumerate(families):
        if i % 2 == 0:
            fig.add_hrect(y0=i - 0.5, y1=i + 0.5, fillcolor="grey", opacity=0.07, line_width=0)

    for fam in families:
        row = fam_to_row[fam]
        for start, end, strand in gt_dict[fam]:
            fig.add_shape(
                type="rect",
                x0=start, x1=end, y0=row - 0.25, y1=row + 0.25,
                fillcolor=color_map[fam], line=dict(color="black", width=0.5),
            )
            # invisible scatter point on top of each bar, for hover tooltip
            fig.add_trace(go.Scatter(
                x=[(start + end) / 2], y=[row],
                mode="markers",
                marker=dict(size=8, color=color_map[fam], symbol="triangle-right" if strand == "+" else "triangle-left"),
                hovertemplate=f"{fam}<br>start={start:,}<br>end={end:,}<br>strand={strand}<extra></extra>",
                showlegend=False
            ))

    fig.update_yaxes(
        tickmode="array", tickvals=list(range(n)), ticktext=families,
        title="IS family"
    )
    fig.update_xaxes(title="Genomic position (bp)", tickformat=",")
    fig.update_layout(
        title="IS element instances by family (zoom/pan to inspect clusters)",
        height=max(400, n * 30),
        width=1200,
        plot_bgcolor="white"
    )
    fig.show()

Combine everything as one fasta file

from Bio import SeqIO
from Bio.Seq import Seq
from Bio.SeqRecord import SeqRecord

# load the reference genome
reference = SeqIO.to_dict(SeqIO.parse("../../data/EC958_complete_genome.fasta", "fasta"))
chrom_seq = reference["HG941718.1"].seq

records = []

for name, locs in gt_dict.items():
    concatenated = Seq("")
    coords_used = []

    for start, end, strand in sorted(locs, key=lambda x: x[0]):
        subseq = chrom_seq[start:end]
        if strand == "-":
            subseq = subseq.reverse_complement()
        concatenated += subseq
        coords_used.append(f"{start}-{end}({strand})")

    record = SeqRecord(
        concatenated,
        id=name,
        description=f"{len(locs)} instance(s): {'; '.join(coords_used)}"
    )
    records.append(record)


SeqIO.write(records, "../../data/gt_IS_sequences.fasta", "fasta")

Analysing it

def match(gt_df, pred_df, buffer_bp):
    """
    gt_df, pred_df: DataFrames with columns ['name', 'start', 'end']
    Returns matched pairs where predicted start/end fall within buffer_bp
    of the ground truth start/end (i.e. "almost the same location").
    """
    matches = []
    used_pred = set()

    for gt_idx, gt in gt_df.iterrows():
        candidates = pred_df[
            (pred_df['chromStart'].between(gt['chromStart'] - buffer_bp, gt['chromStart'] + buffer_bp)) &
            (pred_df['chromEnd'].between(gt['chromEnd'] - buffer_bp, gt['chromEnd'] + buffer_bp))
        ]
        candidates = candidates[~candidates.index.isin(used_pred)]

        if not candidates.empty:
            # pick closest by combined start+end distance
            candidates = candidates.copy()
            candidates['dist'] = (
                (candidates['chromStart'] - gt['chromStart']).abs() +
                (candidates['chromEnd'] - gt['chromEnd']).abs()
            )
            best = candidates.sort_values('dist').iloc[0]
            matches.append({
                'gt_name': gt['name'], 'gt_start': gt['chromStart'], 'gt_end': gt['chromEnd'],
                'pred_name': best['name'], 'pred_start': best['chromStart'], 'pred_end': best['chromEnd'],
                'buffer_bp': buffer_bp
            })
            used_pred.add(best.name)
        else:
            matches.append({
                'gt_name': gt['name'], 'gt_start': gt['chromStart'], 'gt_end': gt['chromEnd'],
                'pred_name': None, 'pred_start': None, 'pred_end': None,
                'buffer_bp': buffer_bp
            })

    return pd.DataFrame(matches)
