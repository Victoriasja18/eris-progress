from collections import defaultdict

def find_location_matches(df1, df2, buffer):
    """
    Stage 1: find every (df2_idx, df1_idx) pair whose location overlaps within buffer.
    df1 = ground truth, df2 = predictions
    Returns a list of candidate matches with both rows' info.
    """
    candidates = []
    for i, pred in df2.iterrows():
        for j, truth in df1.iterrows():
            if pred["transcript"] != truth["transcript"]:
                continue
            if (pred["chromStart"] - buffer) <= truth["chromEnd"] and \
               (pred["chromEnd"]   + buffer) >= truth["chromStart"]:
                candidates.append({"pred_idx": i, "truth_idx": j})
    return candidates

def name_matches(pred_row, truth_name):
    """True if truth_name equals pred's name, or appears in pred's synonyms
    (handles synonyms stored as a real list or as a stringified list)."""
    if pred_row["name"] == truth_name:
        return True
    syns = pred_row.get("synonyms", pd.NA)

    if isinstance(syns, str):
        try:
            syns = ast.literal_eval(syns)
        except (ValueError, SyntaxError):
            syns = []

    if isinstance(syns, list):
        return truth_name in syns
    return False

def precision_recall(df1, df2, buffer):
    candidates = find_location_matches(df1, df2, buffer)

    # group all candidates by which prediction they belong to
    by_pred = defaultdict(list)
    for c in candidates:
        by_pred[c["pred_idx"]].append(c["truth_idx"])

    matched_df1 = set()
    matched_df2 = set()
    wrong_name_records = []

    for i, truth_idxs in by_pred.items():
        pred = df2.loc[i]
        tp_found = False

        # first pass: does ANY candidate match on name?
        for j in truth_idxs:
            truth = df1.loc[j]
            if name_matches(pred, truth["name"]):
                matched_df1.add(j)
                matched_df2.add(i)
                tp_found = True
                break  # one correct match is enough for this prediction

        # only log wrong_name if nothing matched
        if not tp_found:
            j = truth_idxs[0]  # closest/first candidate location
            truth = df1.loc[j]
            record = pred.to_dict()
            record["truth_name"] = truth["name"]
            record["truth_idx"] = j
            record["category"] = "wrong_name"
            wrong_name_records.append(record)

    tp = len(matched_df2)
    wrong_name_pred_idx = {r["truth_idx"] for r in []}  # not used for exclusion below; see fp_idx
    wrong_name_pred_idx = set(pred_fa.loc[[r_i for r_i in by_pred if r_i not in matched_df2]].index)
    fp_idx = set(pred_fa.index) - matched_df2 - wrong_name_pred_idx
    fn = len(df1) - len(matched_df1)
    fp = len(fp_idx)

    precision = tp / (tp + fp) if (tp + fp) > 0 else 0
    recall    = tp / (tp + fn) if (tp + fn) > 0 else 0
    f1        = 2 * precision * recall / (precision + recall) if (precision + recall) > 0 else 0

    summary = {"buffer": buffer, "TP": tp, "FP": fp, "FN": fn,
               "precision": round(precision, 3), "recall": round(recall, 3), "f1": round(f1, 3)}

    tp_df = df2.loc[list(matched_df2)].copy()
    tp_df["category"] = "TP"

    wrong_name_df = pd.DataFrame(wrong_name_records)

    fp_df = df2.loc[list(fp_idx)].copy()
    fp_df["category"] = "FP"

    fn_df = df1.loc[df1.index.difference(matched_df1)].copy()
    fn_df["category"] = "FN"

    detail_df = pd.concat([tp_df, wrong_name_df, fp_df, fn_df], ignore_index=True)
    detail_df["buffer"] = buffer

    return summary, detail_df
