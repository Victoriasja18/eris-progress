# Length and depth of every contig in the stitched path of ISEc8 and ISEc23.
# eris copy number = round(max depth on the path / median depth of all contigs)
from eris.io import GenomeAssembly

genome = GenomeAssembly.from_file(str(mt.GFA))
median_depth = pd.Series(genome.contig_depths).median()
print(f"Median depth of all contigs: {median_depth:.1f}")

for is_name in ["ISEc8", "ISEc23", "IS629"]:
    # match synonyms too (IS629 is reported as IS3411 here); stitched locus first
    loci = read_gfa[read_gfa["target"].map(lambda t: mt.is_group(mt.is_name(t))) == mt.is_group(is_name)]
    locus = loci.sort_values("target", key=lambda t: ~t.str.endswith("_stitched")).iloc[0]
    path = pd.DataFrame({"Contig": locus["locus_contig"].split("|")})
    path["Length"] = path["Contig"].map(lambda c: len(genome[c]))
    path["Depth"] = path["Contig"].map(genome.contig_depths).round(1)
    path["Depth / median"] = (path["Depth"] / median_depth).round(2)
    path[""] = ""
    path.loc[path["Depth"].idxmax(), ""] = "← max"

    print(f"\n{is_name}: eris copy number = {locus['copy_number']:.0f}")
    display(path)
