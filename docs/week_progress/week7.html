combined = None

from Bio import SeqIO
from Bio.Seq import Seq
import re

genome_records = {rec.id: str(rec.seq) for rec in SeqIO.parse("../../data/EC958_complete_genome.fasta", "fasta")}

for rec_id, seq in genome_records.items():
    print(rec_id, len(seq))

def locate_subseq(subseq):
    hits = []
    for rec_id, seq in genome_records.items():
        for m in re.finditer(re.escape(subseq), seq):
            hits.append((rec_id, m.start(), m.end(), '+'))
        rc = str(Seq(subseq).reverse_complement())
        for m in re.finditer(re.escape(rc), seq):
            hits.append((rec_id, m.start(), m.end(), '-'))
    return hits

results = []
for i, row in combined_up_down.iterrows():
    seg_id = str(row['transcript'])
    start, end = int(row['chromStart']), int(row['chromEnd'])
    seg_seq = segments.get(seg_id)
    if seg_seq is None:
        print(f"Warning: segment {seg_id} not found in GFA")
        continue

    subseq = seg_seq[start:end]
    hits = locate_subseq(subseq)

    if len(hits) == 0:
        status = "no_match"
    elif len(hits) == 1:
        status = "unique"
    else:
        status = "ambiguous"  # expected for INSIDE (repeated IS body)

    results.append({
        "locus": row['name'],
        "segment": seg_id,
        "local_start": start,
        "local_end": end,
        "category": row.get('repeats', row.get('category')),
        "genome_hits": hits,
        "status": status,
    })

res_df = pd.DataFrame(results)

count_ambi = 0
for name, grp in res_df.groupby("locus"):
    flanks = grp[grp['category'].isin(['UPSTREAM', 'DOWNSTREAM'])]
    unique_flanks = flanks[flanks['status'] == 'unique']
    if not unique_flanks.empty:
        # pick the flank hit(s) — these tell you the true genome region for this locus
        print(name, unique_flanks[['category','genome_hits']].to_dict('records'))
    else:
        count_ambi += 1
        print(name, "— no unique flank found, needs manual review")

print(f'Total ambiguous is: {count_ambi}')
