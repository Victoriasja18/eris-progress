# Create new columns based on the different postion of IS
def label_is(df, is_col="IS_name", locus_col="locus_id", synonym_lookup=None):
    """
    Adds 'canonical_IS' and 'IS_label' columns.
    Rows with the same IS name AND same locus_id get the same label (e.g. IS1_1).
    A new locus_id under the same canonical IS name gets the next number (IS1_2, IS1_3...).
    """
    df = df.copy()

    # canonicalize synonyms first, if you have a lookup
    if synonym_lookup is not None:
        df["canonical_IS"] = df[is_col].map(lambda x: synonym_lookup.get(x, x))
    else:
        df["canonical_IS"] = df[is_col]

    # rank locus_ids by first appearance, within each canonical_IS group
    def assign_labels(group):
        # order of first appearance of each locus_id within this IS group
        first_seen = group[locus_col].drop_duplicates()
        locus_to_num = {loc: i + 1 for i, loc in enumerate(first_seen)}
        group["IS_label"] = group["canonical_IS"] + "_" + group[locus_col].map(locus_to_num).astype(str)
        return group

    df = df.groupby("canonical_IS", group_keys=False, sort=False).apply(assign_labels)
    return df

import pandas as pd

# Calculate precision and recall
def precision_recall_names(df1, df2, dictionary: dict = None):
    """
    We will do precision and recall based on what IS that found
    df1 = ground truth (reference)
    df2 = predictions to evaluate

    TP = df2 locus that matches a df1 locus within buffer
    FP = df2 locus that has no match in df1
    FN = df1 locus that has no match in df2
    """
    tp, fp, fn = 0, 0, 0
    matched_df1 = set()

    # Drop any duplicates in the predicted dataframe
    gt_labels = set(df1.drop_duplicates(subset=["name"]))
    pred_labels = set(df2.drop_duplicates(subset=["IS_label"]))

    # Calculate the TP, FN, FP
    tp_labels = gt_labels & pred_labels
    fn_labels = gt_labels - pred_labels
    fp_labels = pred_labels - gt_labels

    tp, fp, fn = len(tp_labels), len(fp_labels), len(fn_labels)
    precision = tp / (tp + fp) if (tp + fp) else 0.0
    recall = tp / (tp + fn) if (tp + fn) else 0.0

    summary = {"precision": precision, "recall": recall, "tp": tp, "fp": fp, "fn": fn}

    # actual rows, tagged with their classification
    df2 = df2.copy()
    df2["match_status"] = df2["IS_label"].apply(
        lambda lbl: "TP" if lbl in tp_labels else "FP"
    )
    tp_rows = df2[df2["match_status"] == "TP"]
    fp_rows = df2[df2["match_status"] == "FP"]
    fn_rows = df1[df1['name'].isin(fn_labels)]  # ground truth rows that were missed

    detail = {
        "tp_labels": sorted(tp_labels),
        "fp_labels": sorted(fp_labels),
        "fn_labels": sorted(fn_labels),
        "tp_rows": tp_rows,
        "fp_rows": fp_rows,
        "fn_rows": fn_rows,
        "pred_df_tagged": df2,  # full predicted df with match_status column, if you want to inspect everything at once
    }

    return summary, detail

Recalculate total unique IS

from pathlib import Path

def calc_unique(pathway):
    df = pd.read_table(pathway, sep="\t")
    total_unique = df.groupby("target")["locus_id"].nunique().sum()
    return total_unique
